cmake_minimum_required(VERSION 3.16)
project(xr_device_io VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
find_path(XR_KERNEL_UAPI_DIR linux/a081-nordic.h HINTS /usr/include/xr-kernel)
if(NOT XR_KERNEL_UAPI_DIR)
  message(FATAL_ERROR "Install xr-kernel-uapi-dev or set XR_KERNEL_UAPI_DIR")
endif()
add_library(xr_device_api INTERFACE)
set_target_properties(xr_device_api PROPERTIES EXPORT_NAME api)
target_include_directories(xr_device_api INTERFACE $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<BUILD_INTERFACE:${XR_KERNEL_UAPI_DIR}> $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
add_library(xr_device_abi STATIC src/abi_reference.cpp src/a081_controller_reference.cpp)
set_target_properties(xr_device_abi PROPERTIES POSITION_INDEPENDENT_CODE ON EXPORT_NAME abi)
target_link_libraries(xr_device_abi PUBLIC xr_device_api)
add_executable(xr-nordic-stream src/nordic-stream.cpp)
target_link_libraries(xr-nordic-stream PRIVATE xr_device_abi pthread)
add_executable(xr-controller-radio src/controller-radio.cpp)
target_link_libraries(xr-controller-radio PRIVATE xr_device_api pthread)
install(TARGETS xr-nordic-stream xr-controller-radio RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(TARGETS xr_device_abi xr_device_api EXPORT XRDeviceTargets ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(DIRECTORY include/xr DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(EXPORT XRDeviceTargets NAMESPACE XRDevice:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/XRDevice)
configure_package_config_file(cmake/XRDeviceConfig.cmake.in ${CMAKE_CURRENT_BINARY_DIR}/XRDeviceConfig.cmake INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/XRDevice)
write_basic_package_version_file(${CMAKE_CURRENT_BINARY_DIR}/XRDeviceConfigVersion.cmake VERSION ${PROJECT_VERSION} COMPATIBILITY SameMajorVersion)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/XRDeviceConfig.cmake ${CMAKE_CURRENT_BINARY_DIR}/XRDeviceConfigVersion.cmake DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/XRDevice)
